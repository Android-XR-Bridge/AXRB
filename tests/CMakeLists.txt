function(axrb_native_test target source test_name)
  add_executable(${target} native/${source})
  target_link_libraries(${target} PRIVATE axrb_protocol)
  # Smoke tests use assert; keep checks active in Release builds too.
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ${test_name} COMMAND ${target})
endfunction()
